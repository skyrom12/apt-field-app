// ---------- Tech view ----------

function apiGetRoute(tech, day) {
  if (TECHS.indexOf(tech) < 0) throw new Error('Unknown tech: ' + tech);
  if (DAYS.indexOf(day) < 0) throw new Error('Unknown day: ' + day);

  const stops = readTable_('Routes')
    .filter(function (r) { return r.Tech === tech && r.Day === day; })
    .sort(function (a, b) { return a.Order - b.Order; });

  // Mark stops already logged today so techs see progress.
  const todayKey = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd');
  const doneToday = {};
  readTable_('VisitLog').forEach(function (v) {
    const vDate = Utilities.formatDate(new Date(v.Timestamp), Session.getScriptTimeZone(), 'yyyy-MM-dd');
    if (vDate === todayKey && v.Status !== 'Rejected') doneToday[v.Customer + '|' + v.Address] = v.Status;
  });

  return stops.map(function (s) {
    return {
      order: s.Order, customer: s.Customer, address: s.Address,
      lat: s.Lat, lng: s.Lng,
      status: doneToday[s.Customer + '|' + s.Address] || null
    };
  });
}

// ---------- Prices ----------
// Source of truth: the Prices tab. Office edits prices there — no deploy needed.
// Keys are stable ids; the future QuickBooks line-item automation maps them to
// QBO service items (see field-app.md).

function apiGetPrices() {
  return { items: getPrices_() };
}

function getPrices_() {
  return readTable_('Prices').map(function (p) {
    return {
      key: String(p.Key), label: String(p.Label), unit: String(p.Unit),
      price: Math.round((Number(p.Price) || 0) * 100) / 100
    };
  });
}

/**
 * payload: {tech, day, customer, address, services:[], noteToCustomer,
 *           internalNote, skipped, skipReason, photos:[{name,mimeType,dataBase64}],
 *           workOrder: {                       // digitized paper work order
 *             timeIn, timeOut,                 // ISO strings (optional)
 *             water: {arriveLevel, leftLevel, pressureIn, pressureOut, airTemp, waterTemp},
 *             readings: {ch, tc, fc, ph, alk, cya, salt, phos},
 *             chems: {priceKey: qty, ...},     // nonzero quantities only
 *             equipment: {name: 'E'|'L'|'M'|'H', ...},
 *             filters: {cleaned: 'Y'|'N'|null, amount: 'L'|'M'|'H'|null},
 *             repairs: '', custChems: bool}}
 * Legacy payloads with {readings:{pH,..}, chemicals:'text'} still work.
 */
function apiSubmitVisit(payload) {
  if (TECHS.indexOf(payload.tech) < 0) throw new Error('Unknown tech.');
  const visitId = 'V' + Date.now() + '-' + Math.floor(Math.random() * 1000);
  const now = new Date();

  const photoLinks = (payload.photos || []).map(function (p, i) {
    const folder = customerPhotoFolder_(payload.customer);
    const stamp = Utilities.formatDate(now, Session.getScriptTimeZone(), 'yyyy-MM-dd');
    const blob = Utilities.newBlob(
      Utilities.base64Decode(p.dataBase64), p.mimeType || 'image/jpeg',
      stamp + ' ' + payload.customer + ' ' + (i + 1) + '.jpg');
    const file = folder.createFile(blob);
    return file.getUrl();
  });

  const status = payload.skipped ? 'Skipped' : 'Pending';
  const services = payload.skipped ? ('UNABLE TO SERVICE: ' + (payload.skipReason || ''))
                                   : (payload.services || []).join(', ');

  const wo = payload.workOrder || null;
  let readings, chemicals = payload.chemicals || '', chemCharge = 0;

  if (wo) {
    readings = woReadingsSummary_(wo);
    // Chemical charge is computed HERE, from the Prices tab, never trusted
    // from the client. Rounded per line, then summed.
    const priceMap = {};
    getPrices_().forEach(function (p) { priceMap[p.key] = p; });
    const lines = [];
    Object.keys(wo.chems || {}).forEach(function (k) {
      const qty = Number(wo.chems[k]) || 0;
      if (qty <= 0) return;
      const p = priceMap[k];
      if (p) {
        chemCharge += Math.round(qty * p.price * 100) / 100;
        lines.push(qty + ' ' + p.unit + ' ' + p.label);
      } else {
        lines.push(qty + ' ' + k + ' (unknown item — not priced)');
      }
    });
    chemCharge = Math.round(chemCharge * 100) / 100;
    if (wo.custChems) chemCharge = 0;
    chemicals = lines.join(', ') + (wo.custChems && lines.length ? ' — customer\'s own chemicals, no charge' : '');
  } else {
    readings = payload.readings
      ? Object.keys(payload.readings)
          .filter(function (k) { return payload.readings[k] !== '' && payload.readings[k] != null; })
          .map(function (k) { return k + ': ' + payload.readings[k]; }).join(', ')
      : '';
  }

  getSheet_('VisitLog').appendRow([
    visitId, now, payload.tech, payload.day, payload.customer, payload.address,
    services, readings, chemicals, payload.noteToCustomer || '',
    payload.internalNote || '', photoLinks.join('\n'), status, '', '', '', '', '',
    wo ? JSON.stringify(wo) : '', chemCharge
  ]);

  return { visitId: visitId, status: status, photos: photoLinks.length, chemCharge: chemCharge };
}

/** Human-readable summary of work-order readings for the Readings column/email. */
function woReadingsSummary_(wo) {
  const r = wo.readings || {}, w = wo.water || {};
  const labels = { ch: 'CH', tc: 'TC', fc: 'FC', ph: 'pH', alk: 'Alk', cya: 'CYA', salt: 'Salt', phos: 'Phosphates' };
  const parts = [];
  Object.keys(labels).forEach(function (k) {
    if (r[k] !== '' && r[k] != null) parts.push(labels[k] + ': ' + r[k]);
  });
  if (w.arriveLevel) parts.push('Level in/out: ' + w.arriveLevel + '/' + (w.leftLevel || '-'));
  if (w.pressureIn || w.pressureOut) parts.push('PSI in/out: ' + (w.pressureIn || '-') + '/' + (w.pressureOut || '-'));
  if (w.airTemp) parts.push('Air: ' + w.airTemp + '°');
  if (w.waterTemp) parts.push('Water: ' + w.waterTemp + '°');
  return parts.join(', ');
}

function customerPhotoFolder_(customer) {
  const root = DriveApp.getFolderById(PROPS.getProperty('PHOTOS_FOLDER_ID'));
  const safe = customer.replace(/[\\/:*?"<>|]/g, '-');
  const it = root.getFoldersByName(safe);
  return it.hasNext() ? it.next() : root.createFolder(safe);
}

// ---------- Office view ----------

function apiOfficeState(pin) {
  requirePin_(pin);
  const custByKey = {};
  readTable_('Customers').forEach(function (c) {
    custByKey[c.Customer + '|' + c.Address] = c;
  });
  const pending = readTable_('VisitLog').filter(function (v) {
    return v.Status === 'Pending' || v.Status === 'Skipped' || v.Status === 'Error';
  });
  return {
    qbConnected: qbIsConnected(),
    qbAuthUrl: qbIsConnected() ? null : qbGetAuthorizationUrl(),
    pending: pending.map(function (v) {
      const cust = custByKey[v.Customer + '|' + v.Address];
      let wo = null;
      try { wo = v.WorkOrder ? JSON.parse(v.WorkOrder) : null; } catch (e) {}
      return {
        visitId: v.VisitId,
        when: Utilities.formatDate(new Date(v.Timestamp), Session.getScriptTimeZone(), 'EEE MMM d h:mm a'),
        tech: v.Tech, customer: v.Customer, address: v.Address,
        services: v.Services, readings: v.Readings, chemicals: v.Chemicals,
        noteToCustomer: v.NoteToCustomer, internalNote: v.InternalNote,
        photos: v.Photos ? v.Photos.split('\n') : [],
        status: v.Status, error: v.Error,
        workOrder: wo,
        chemCharge: Math.round((Number(v.ChemCharge) || 0) * 100) / 100,
        rate: cust ? (Number(cust.Rate) || 0) : 0
      };
    })
  };
}

/**
 * Approve a visit. options: {invoice:bool, email:bool}
 * Charges/invoices happen ONLY here — never on tech submission.
 * FUTURE (billing automation): qbCreateInvoice currently bills the flat Rate;
 * when we automate chemical billing, add one QBO line per entry in the visit's
 * WorkOrder.chems using the Prices tab keys → QBO items. ChemCharge already
 * holds the computed total, so the invoice amount is Rate + ChemCharge.
 */
function apiApproveVisit(pin, visitId, options) {
  requirePin_(pin);
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const sh = getSheet_('VisitLog');
    const visits = readTable_('VisitLog');
    const v = visits.filter(function (x) { return x.VisitId === visitId; })[0];
    if (!v) throw new Error('Visit not found: ' + visitId);
    if (v.Status === 'Approved') throw new Error('Already approved.');

    const customer = readTable_('Customers').filter(function (c) {
      return c.Customer === v.Customer && c.Address === v.Address;
    })[0];

    const result = { invoiceId: null, emailed: false, warnings: [] };

    if (options.invoice) {
      if (!customer) throw new Error('Customer not found in Customers tab.');
      if (!customer.Rate) throw new Error('No Rate set for ' + v.Customer + ' — fill the Rate column in the Customers tab.');
      result.invoiceId = qbCreateInvoice(v, customer);
      sh.getRange(v._row, 14).setValue(result.invoiceId); // InvoiceId
    }

    if (options.email) {
      if (!customer || !customer.Email) {
        result.warnings.push('No email on file for ' + v.Customer + ' — confirmation not sent.');
      } else {
        sendVisitEmail(v, customer);
        sh.getRange(v._row, 17).setValue(new Date()); // EmailSentAt
        result.emailed = true;
      }
    }

    sh.getRange(v._row, 13).setValue('Approved');   // Status
    sh.getRange(v._row, 15).setValue('office');     // ApprovedBy
    sh.getRange(v._row, 16).setValue(new Date());   // ApprovedAt
    sh.getRange(v._row, 18).setValue('');           // clear Error
    return result;
  } catch (err) {
    markVisitError_(visitId, String(err.message || err));
    throw err;
  } finally {
    lock.releaseLock();
  }
}

function apiRejectVisit(pin, visitId, reason) {
  requirePin_(pin);
  const v = readTable_('VisitLog').filter(function (x) { return x.VisitId === visitId; })[0];
  if (!v) throw new Error('Visit not found.');
  const sh = getSheet_('VisitLog');
  sh.getRange(v._row, 13).setValue('Rejected');
  sh.getRange(v._row, 18).setValue(reason || '');
  return true;
}

function markVisitError_(visitId, msg) {
  try {
    const v = readTable_('VisitLog').filter(function (x) { return x.VisitId === visitId; })[0];
    if (v) {
      const sh = getSheet_('VisitLog');
      sh.getRange(v._row, 13).setValue('Error');
      sh.getRange(v._row, 18).setValue(msg);
    }
  } catch (e) { /* best effort */ }
}
/**
 * APT Field App — customer management.
 *
 * The office view's Customers panel: list who is on the routes, fix a rate or an
 * email, and take someone off the routes. Everything here is office-PIN gated and
 * runs under the script lock, because all of it writes.
 *
 *   apiListCustomers(pin)                        → every customer, their route
 *                                                  days, rate/email, and whether
 *                                                  visits of theirs are waiting
 *   apiUpdateCustomer(pin, customer, address, f) → patch rate/email/phone/QBO id
 *   apiRemoveCustomer(pin, customer, address)    → delete their Routes rows,
 *                                                  renumber the days they were on,
 *                                                  mark the customer inactive
 *
 * Removal never deletes the Customers row and never touches VisitLog: history and
 * photos survive, and a customer who comes back is re-added through the normal
 * "Add a customer" flow, which re-places them optimally against the routes as
 * they stand by then.
 */

/** Statuses that mean a visit is still waiting on the office. */
const OPEN_VISIT_STATUSES = ['Pending', 'Skipped', 'Error'];

function custKey_(customer, address) {
  return String(customer || '').trim() + '|' + String(address || '').trim();
}

/** Column number (1-based) of a header on a sheet, or 0. */
function colOf_(sh, header) {
  const head = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0];
  return head.indexOf(header) + 1;
}

/**
 * Every customer with their route days and anything the office needs to act on.
 * Also reports orphan route stops — Routes rows with no matching Customers row,
 * which bill at $0 and email nobody, so they are worth surfacing rather than
 * leaving to be discovered at invoice time.
 */
function apiListCustomers(pin) {
  requirePin_(pin);

  const groups = routeGroups_();
  const stopsByCustomer = {};
  Object.keys(groups).forEach(function (k) {
    groups[k].stops.forEach(function (s) {
      const key = custKey_(s.customer, s.address);
      if (!stopsByCustomer[key]) stopsByCustomer[key] = [];
      stopsByCustomer[key].push({ tech: groups[k].tech, day: groups[k].day, order: s.order });
    });
  });

  const openVisits = {};
  readTable_('VisitLog').forEach(function (v) {
    if (OPEN_VISIT_STATUSES.indexOf(String(v.Status)) < 0) return;
    const key = custKey_(v.Customer, v.Address);
    openVisits[key] = (openVisits[key] || 0) + 1;
  });

  const seen = {};
  const customers = readTable_('Customers').map(function (c) {
    const customer = String(c.Customer || '').trim();
    const address = String(c.Address || '').trim();
    const key = custKey_(customer, address);
    seen[key] = true;
    const days = (stopsByCustomer[key] || []).sort(function (a, b) {
      const d = DAYS.indexOf(a.day) - DAYS.indexOf(b.day);
      return d !== 0 ? d : a.tech.localeCompare(b.tech);
    });
    const rate = Number(c.Rate);
    return {
      customer: customer,
      address: address,
      email: String(c.Email || '').trim(),
      phone: String(c.Phone || '').trim(),
      rate: isFinite(rate) && c.Rate !== '' ? rate : null,
      qbCustomerId: String(c.QBCustomerId || '').trim(),
      // Active is a label the office sets; it does not by itself stop a visit.
      // Removing the Routes rows is what does that, and this is set with it.
      active: String(c.Active || '').toUpperCase() !== 'FALSE',
      days: days,
      visits: days.length,
      openVisits: openVisits[key] || 0
    };
  });

  // Route stops with no Customers row: they bill at $0 and email nobody.
  const orphans = [];
  Object.keys(stopsByCustomer).forEach(function (key) {
    if (seen[key]) return;
    const parts = key.split('|');
    orphans.push({ customer: parts[0], address: parts[1], days: stopsByCustomer[key] });
  });

  customers.sort(function (a, b) { return a.customer.localeCompare(b.customer); });
  orphans.sort(function (a, b) { return a.customer.localeCompare(b.customer); });
  return { customers: customers, orphans: orphans };
}

/**
 * Patch a customer's billing details. Only the fields passed are written, so the
 * office can fix a rate without clearing an email.
 *
 * fields: {email, phone, rate, qbCustomerId} — any subset. rate '' clears it.
 */
function apiUpdateCustomer(pin, customer, address, fields) {
  requirePin_(pin);
  const f = fields || {};
  const key = custKey_(customer, address);

  let rate;
  if (f.rate !== undefined) {
    if (f.rate === '' || f.rate === null) {
      rate = '';
    } else {
      rate = Number(f.rate);
      if (!isFinite(rate) || rate < 0) throw new Error('Rate must be a number (no $ sign).');
    }
  }

  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const sh = getSheet_('Customers');
    const row = readTable_('Customers').filter(function (c) {
      return custKey_(c.Customer, c.Address) === key;
    })[0];
    if (!row) throw new Error('No Customers row for ' + customer + '.');

    const writes = [];
    if (f.email !== undefined) writes.push(['Email', String(f.email).trim()]);
    if (f.phone !== undefined) writes.push(['Phone', String(f.phone).trim()]);
    if (f.qbCustomerId !== undefined) writes.push(['QBCustomerId', String(f.qbCustomerId).trim()]);
    if (rate !== undefined) writes.push(['Rate', rate]);

    writes.forEach(function (w) {
      const col = colOf_(sh, w[0]);
      if (col) sh.getRange(row._row, col).setValue(w[1]);
    });

    return { updated: writes.length };
  } finally {
    lock.releaseLock();
  }
}

/**
 * Take a customer off the routes.
 *
 * Deletes every Routes row for them (a twice-weekly customer has two), renumbers
 * each day they were on so the techs' stop numbers stay 1…N, and marks the
 * Customers row inactive. The Customers row itself stays: VisitLog references it
 * by name and address, and deleting it would break approval of any visit of
 * theirs still in the queue.
 *
 * Refuses while they have visits waiting, rather than leaving the office with a
 * queue entry it can no longer invoice.
 */
function apiRemoveCustomer(pin, customer, address) {
  requirePin_(pin);
  const name = String(customer || '').trim();
  const addr = String(address || '').trim();
  if (!name || !addr) throw new Error('Customer and address are required.');
  const key = custKey_(name, addr);

  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const open = readTable_('VisitLog').filter(function (v) {
      return custKey_(v.Customer, v.Address) === key &&
             OPEN_VISIT_STATUSES.indexOf(String(v.Status)) >= 0;
    });
    if (open.length) {
      throw new Error(name + ' has ' + open.length + ' visit' + (open.length > 1 ? 's' : '') +
        ' still waiting in the queue. Approve or reject ' +
        (open.length > 1 ? 'those' : 'that') + ' first, then remove them.');
    }

    const groups = routeGroups_();
    const hits = [];
    Object.keys(groups).forEach(function (k) {
      groups[k].stops.forEach(function (s) {
        if (custKey_(s.customer, s.address) === key) {
          hits.push({ row: s.row, tech: groups[k].tech, day: groups[k].day });
        }
      });
    });
    if (!hits.length) throw new Error(name + ' is not on any route.');

    // Delete bottom-up: every deletion shifts the rows below it.
    const sh = getSheet_('Routes');
    hits.slice().sort(function (a, b) { return b.row - a.row; })
        .forEach(function (h) { sh.deleteRow(h.row); });
    SpreadsheetApp.flush();

    // Renumber each day they were on, so no gaps reach the techs.
    const touched = {};
    hits.forEach(function (h) { touched[h.tech + '|' + h.day] = h; });
    Object.keys(touched).forEach(function (k) {
      renumberDay_(touched[k].tech, touched[k].day, null, null);
    });

    // Mark inactive; keep the row so history and any later re-add still resolve.
    const custSheet = getSheet_('Customers');
    const custRow = readTable_('Customers').filter(function (c) {
      return custKey_(c.Customer, c.Address) === key;
    })[0];
    let flagged = false;
    if (custRow) {
      const col = colOf_(custSheet, 'Active');
      if (col) { custSheet.getRange(custRow._row, col).setValue('FALSE'); flagged = true; }
    }

    return {
      removed: hits.length,
      days: hits.map(function (h) { return h.tech + ' ' + h.day; }),
      markedInactive: flagged
    };
  } finally {
    lock.releaseLock();
  }
}
/**
 * APT Field App — route placement.
 *
 * Lets the office add a customer without anyone re-running the optimizer by
 * hand. Two office-PIN-gated steps:
 *
 *   apiSuggestPlacement(pin, address) → geocodes the address and ranks every
 *     tech/day slot: "Adam · Wed · between Ladart, Anna and Bishop, Lacey — +1.2 mi"
 *   apiAddCustomer(pin, payload)      → writes the Routes + Customers rows and
 *     renumbers that day's Order column 1…N
 *
 * Method: cheapest insertion on straight-line distance — the same measure the
 * 2026-08-21 optimization was scored on, so "+1.2 mi" here is comparable to
 * that plan's between-stop miles. Road miles run higher; the ranking holds.
 *
 * Scope: this places ONE stop well against the routes as they stand. It does
 * not re-solve the week. Twenty good local insertions still drift from a true
 * optimum — see ADDING-CUSTOMERS.md for when to ask DIDM for a full re-run.
 */

/**
 * Tunables. These are judgment calls, not physics — edit them here if the
 * office's experience says otherwise.
 */
const PLACEMENT = {
  // A day at or above this many stops gets penalized, not banned: a stop that
  // is genuinely next door still wins over a long detour on a lighter day.
  maxStopsPerDay: 11,
  // Miles of "cost" added per stop above the cap.
  overloadPenaltyMi: 4,
  // Cost charged for opening a tech's empty day for one stop. Between-stop
  // miles don't see the drive out and back, so an empty day would otherwise
  // score 0 and win everything. Keeps Zack's open Mon–Thu available without
  // letting them swallow every new customer.
  newDayCostMi: 30,
  // How many slots to offer the office.
  suggestions: 3,
  // Geocoder bias box: the whole service area — DeSoto County MS in the south
  // up through Tipton County TN in the north, Fisherville/Oakland in the east.
  // Stops "1st St" from resolving to the wrong state. Margin is deliberate: an
  // earlier 35.45 northern edge clipped Brighton (~35.48) and Covington
  // (~35.57), both squarely in Nick's territory.
  bounds: { swLat: 34.70, swLng: -90.35, neLat: 35.70, neLng: -89.30 }
};

// ---------- geometry ----------

/** Great-circle miles between two {lat, lng} points. */
function milesBetween_(a, b) {
  const R = 3958.8;
  const rad = Math.PI / 180;
  const dLat = (b.lat - a.lat) * rad;
  const dLng = (b.lng - a.lng) * rad;
  const h = Math.sin(dLat / 2) * Math.sin(dLat / 2) +
            Math.cos(a.lat * rad) * Math.cos(b.lat * rad) *
            Math.sin(dLng / 2) * Math.sin(dLng / 2);
  return 2 * R * Math.asin(Math.min(1, Math.sqrt(h)));
}

function hasCoords_(s) {
  return isFinite(s.lat) && isFinite(s.lng) && !(s.lat === 0 && s.lng === 0);
}

function round1_(n) { return Math.round(n * 10) / 10; }

/**
 * Cheapest place to slot `stop` into the ordered array `seq`.
 *
 * Returns {index, detourMi, after, before} — `after`/`before` are the stops the
 * new one lands between (null at either end). Routes are open paths (techs
 * don't return to a depot), so inserting at an end costs only the one new leg.
 */
function bestInsertion_(seq, stop) {
  if (!seq.length) return { index: 0, detourMi: 0, after: null, before: null };
  let best = null;
  for (let i = 0; i <= seq.length; i++) {
    const after = i > 0 ? seq[i - 1] : null;      // stop the new one follows
    const before = i < seq.length ? seq[i] : null; // stop the new one precedes
    let detour;
    if (!after) detour = milesBetween_(stop, before);
    else if (!before) detour = milesBetween_(after, stop);
    else detour = milesBetween_(after, stop) + milesBetween_(stop, before) -
                  milesBetween_(after, before);
    // `<=` keeps the LATEST slot among equal-cost ones. Ties are real (a
    // one-stop day costs the same leg either side), and inserting later shifts
    // fewer of the numbers the techs already know.
    if (!best || detour <= best.detourMi) {
      best = { index: i, detourMi: detour, after: after, before: before };
    }
  }
  return best;
}

/**
 * An Order value that sorts into the chosen slot. Exact value doesn't matter —
 * the day is renumbered 1…N right after the write — it only has to land
 * between its neighbours.
 */
function slotOrder_(after, before) {
  if (!after && !before) return 1;
  if (!after) return Number(before.order) - 0.5;
  if (!before) return Number(after.order) + 0.5;
  return (Number(after.order) + Number(before.order)) / 2;
}

// ---------- reading the routes ----------

/** Routes grouped by "Tech|Day", each stops[] sorted by Order. */
function routeGroups_() {
  const groups = {};
  readTable_('Routes').forEach(function (r) {
    const tech = String(r.Tech || '').trim();
    const day = String(r.Day || '').trim();
    if (!tech || !day) return;
    const key = tech + '|' + day;
    if (!groups[key]) groups[key] = { tech: tech, day: day, stops: [] };
    groups[key].stops.push({
      row: r._row,
      order: Number(r.Order),
      customer: String(r.Customer || ''),
      address: String(r.Address || ''),
      lat: Number(r.Lat),
      lng: Number(r.Lng)
    });
  });
  Object.keys(groups).forEach(function (k) {
    groups[k].stops.sort(function (a, b) { return a.order - b.order; });
  });
  return groups;
}

/**
 * Rank every tech/day slot for a new stop, cheapest first. Empty tech/day
 * combinations are included at newDayCostMi so open capacity (Zack Mon–Thu)
 * stays visible without winning by default.
 *
 * Stops missing coordinates are skipped for the geometry but still counted for
 * load, so a hand-added row without lat/lng degrades the estimate rather than
 * breaking it.
 */
function rankPlacements_(groups, stop) {
  const all = {};
  Object.keys(groups).forEach(function (k) { all[k] = groups[k]; });
  TECHS.forEach(function (tech) {
    DAYS.forEach(function (day) {
      const key = tech + '|' + day;
      if (!all[key]) all[key] = { tech: tech, day: day, stops: [] };
    });
  });

  const ranked = Object.keys(all).map(function (key) {
    const g = all[key];
    const seq = g.stops.filter(hasCoords_);
    const ins = bestInsertion_(seq, stop);
    const load = g.stops.length;
    const detourMi = seq.length ? ins.detourMi : PLACEMENT.newDayCostMi;
    const overBy = Math.max(0, (load + 1) - PLACEMENT.maxStopsPerDay);
    const penaltyMi = overBy * PLACEMENT.overloadPenaltyMi;
    return {
      tech: g.tech,
      day: g.day,
      load: load,
      newDay: seq.length === 0,
      detourMi: round1_(detourMi),
      penaltyMi: round1_(penaltyMi),
      scoreMi: round1_(detourMi + penaltyMi),
      afterCustomer: ins.after ? ins.after.customer : null,
      beforeCustomer: ins.before ? ins.before.customer : null,
      newOrder: slotOrder_(ins.after, ins.before),
      // Where the stop lands in the day once renumbered, for the office's sake.
      position: ins.index + 1
    };
  });

  ranked.sort(function (a, b) {
    if (a.scoreMi !== b.scoreMi) return a.scoreMi - b.scoreMi;
    return a.load - b.load; // tie → the lighter day
  });
  return ranked;
}

// ---------- geocoding ----------

/**
 * Address → coordinates via the built-in Maps service (no API key, ~1k/day).
 * If this ever throws an authorization error after a scope change, clear the
 * oauthScopes block in appsscript.json, re-run setup() to let Apps Script
 * recompute scopes, and re-authorize.
 */
function geocodeAddress_(address) {
  const b = PLACEMENT.bounds;
  const res = Maps.newGeocoder()
    .setRegion('us')
    .setBounds(b.swLat, b.swLng, b.neLat, b.neLng)
    .geocode(address);

  if (res.status !== 'OK' || !res.results || !res.results.length) {
    throw new Error('Could not find "' + address + '" (' + res.status +
      '). Add the city, state and ZIP, or type the coordinates in by hand.');
  }
  const top = res.results[0];
  const loc = top.geometry.location;
  return {
    lat: loc.lat,
    lng: loc.lng,
    formattedAddress: top.formatted_address,
    // ROOFTOP/RANGE_INTERPOLATED are trustworthy; APPROXIMATE usually means it
    // only found the street or the ZIP centroid.
    approximate: top.geometry.location_type === 'APPROXIMATE' || !!top.partial_match,
    matchCount: res.results.length
  };
}

// ---------- API ----------

/**
 * Step 1: work out where the stop should go. Nothing is written.
 *
 * Normally geocodes the address. Pass lat/lng to skip that — the office needs
 * a way through when the geocoder can't find a new subdivision or a private
 * drive, by right-clicking the pin in Google Maps and copying the coordinates.
 */
function apiSuggestPlacement(pin, address, lat, lng) {
  requirePin_(pin);
  const q = String(address || '').trim();
  if (!q) throw new Error('Enter the address first.');

  const manual = { lat: Number(lat), lng: Number(lng) };
  const geo = hasCoords_(manual)
    ? { lat: manual.lat, lng: manual.lng, formattedAddress: q,
        approximate: false, matchCount: 1, manual: true }
    : geocodeAddress_(q);
  const ranked = rankPlacements_(routeGroups_(), { lat: geo.lat, lng: geo.lng });

  return {
    lat: geo.lat,
    lng: geo.lng,
    formattedAddress: geo.formattedAddress,
    approximate: geo.approximate,
    matchCount: geo.matchCount,
    manual: !!geo.manual,
    suggestions: ranked.slice(0, PLACEMENT.suggestions)
  };
}

/**
 * Step 2: write the customer in.
 *
 * payload: {customer, address, lat, lng, tech, day, newOrder,
 *           email, phone, rate, qbCustomerId}
 *
 * Writes the Routes row, renumbers that tech/day 1…N, and adds the Customers
 * row if the pair isn't already there. Returns warnings rather than failing
 * when Rate or Email are missing — the office often fills those in later.
 */
function apiAddCustomer(pin, payload) {
  requirePin_(pin);
  const p = payload || {};
  const customer = String(p.customer || '').trim();
  const address = String(p.address || '').trim();
  const tech = String(p.tech || '').trim();
  const day = String(p.day || '').trim();
  const lat = Number(p.lat);
  const lng = Number(p.lng);

  if (!customer) throw new Error('Customer name is required.');
  if (!address) throw new Error('Address is required.');
  if (TECHS.indexOf(tech) < 0) throw new Error('Unknown tech: ' + tech);
  if (DAYS.indexOf(day) < 0) throw new Error('Unknown day: ' + day);
  if (!hasCoords_({ lat: lat, lng: lng })) {
    throw new Error('No coordinates — run "Find the best spot" first, or type the lat/lng in by hand.');
  }

  let rate = '';
  if (p.rate !== '' && p.rate != null) {
    rate = Number(p.rate);
    if (!isFinite(rate) || rate < 0) throw new Error('Rate must be a number (no $ sign).');
  }

  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const warnings = [];

    // A repeat of the same name+address on the same day is a double-submit.
    // The same pair on a DIFFERENT day is legitimate — that's how a
    // twice-weekly customer like Bishop, Lacey is set up.
    const groups = routeGroups_();
    const existing = groups[tech + '|' + day];
    if (existing && existing.stops.some(function (s) {
      return s.customer.trim() === customer && s.address.trim() === address;
    })) {
      throw new Error(customer + ' is already on ' + tech + '’s ' + day + ' route.');
    }
    const otherDays = [];
    Object.keys(groups).forEach(function (k) {
      groups[k].stops.forEach(function (s) {
        if (s.customer.trim() === customer && s.address.trim() === address) {
          otherDays.push(groups[k].tech + ' ' + groups[k].day);
        }
      });
    });
    if (otherDays.length) {
      warnings.push('Already on ' + otherDays.join(', ') +
        ' — this adds a second visit each week. Remove the other row if that is not intended.');
    }

    const routes = getSheet_('Routes');
    const rHead = routes.getRange(1, 1, 1, routes.getLastColumn()).getValues()[0];
    const rowValues = rHead.map(function (h) {
      switch (h) {
        case 'Tech': return tech;
        case 'Day': return day;
        case 'Order': return Number(p.newOrder) || 999;
        case 'Customer': return customer;
        case 'Address': return address;
        case 'Lat': return lat;
        case 'Lng': return lng;
        default: return '';
      }
    });
    routes.appendRow(rowValues);
    SpreadsheetApp.flush();

    const finalOrder = renumberDay_(tech, day, customer, address);

    // Customers tab: one row per name+address, however many days they run.
    const custSheet = getSheet_('Customers');
    const already = readTable_('Customers').some(function (c) {
      return String(c.Customer).trim() === customer && String(c.Address).trim() === address;
    });
    if (!already) {
      const cHead = custSheet.getRange(1, 1, 1, custSheet.getLastColumn()).getValues()[0];
      custSheet.appendRow(cHead.map(function (h) {
        switch (h) {
          case 'Customer': return customer;
          case 'Address': return address;
          case 'Email': return String(p.email || '').trim();
          case 'Phone': return String(p.phone || '').trim();
          case 'Rate': return rate;
          case 'QBCustomerId': return String(p.qbCustomerId || '').trim();
          case 'Active': return 'TRUE';
          default: return '';
        }
      }));
    } else {
      // Coming back after a removal: flip them active again, leave the rest alone.
      const existingRow = readTable_('Customers').filter(function (c) {
        return String(c.Customer).trim() === customer && String(c.Address).trim() === address;
      })[0];
      const activeCol = colOf_(custSheet, 'Active');
      if (existingRow && activeCol) custSheet.getRange(existingRow._row, activeCol).setValue('TRUE');
      warnings.push('Customer already had a Customers row — left its email and rate alone.');
    }

    if (!already && !String(p.email || '').trim()) {
      warnings.push('No email — this customer will not get visit confirmations until you add one.');
    }
    if (!already && rate === '') {
      warnings.push('No rate — invoicing this customer will fail until you set one.');
    }

    return { tech: tech, day: day, order: finalOrder, warnings: warnings };
  } finally {
    lock.releaseLock();
  }
}

/**
 * Smoke test — run this from the Apps Script editor right after pasting
 * Routing.gs in, BEFORE cutting a deployment. Geocodes one address against the
 * real Routes tab and logs the ranked slots. Writes nothing.
 *
 * First run will prompt to authorize the Maps service — accept it. If the log
 * shows a sane tech/day for a Germantown address, placement works on real data.
 * Change `address` to test a specific new customer.
 */
function smokeTestPlacement() {
  const address = '9155 Poplar Ave, Germantown, TN 38138';

  const geo = geocodeAddress_(address);
  Logger.log('Geocoded: ' + geo.formattedAddress +
    ' (' + geo.lat + ', ' + geo.lng + ')' + (geo.approximate ? '  [APPROXIMATE]' : ''));

  const groups = routeGroups_();
  const rows = Object.keys(groups).reduce(function (n, k) { return n + groups[k].stops.length; }, 0);
  Logger.log('Routes tab: ' + rows + ' stops across ' + Object.keys(groups).length + ' tech/days');

  const ranked = rankPlacements_(groups, { lat: geo.lat, lng: geo.lng });
  ranked.slice(0, 5).forEach(function (s, i) {
    Logger.log((i + 1) + '. ' + s.tech + ' ' + s.day +
      ' — stop ' + s.position + ' of ' + (s.load + 1) +
      ' — +' + s.detourMi + ' mi' + (s.penaltyMi ? ' (+' + s.penaltyMi + ' overload)' : '') +
      ' — ' + (s.newDay ? 'opens an empty day'
        : 'between ' + (s.afterCustomer || 'the start') + ' and ' + (s.beforeCustomer || 'the end')));
  });
  return ranked.slice(0, 5);
}

/**
 * Rewrite one tech/day's Order column as 1…N in current sort order, so the
 * fractional value used to slot a new stop never reaches the techs. Returns
 * the new whole-number Order of the named customer.
 */
function renumberDay_(tech, day, customer, address) {
  const sh = getSheet_('Routes');
  const head = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0];
  const orderCol = head.indexOf('Order') + 1;
  if (!orderCol) throw new Error('Routes tab has no Order column.');

  const stops = (routeGroups_()[tech + '|' + day] || { stops: [] }).stops;
  let placed = null;
  stops.forEach(function (s, i) {
    const n = i + 1;
    if (s.order !== n) sh.getRange(s.row, orderCol).setValue(n);
    if (s.customer.trim() === customer && s.address.trim() === address) placed = n;
  });
  return placed;
}
